   292→            emailHref: `mailto:${data.savEmail}`,

   294→        }
   295→        if (data.savHoraires) setSavHoraires(data.savHoraires);
   296→        if (data.savFaq) setSavFaq(data.savFaq);
   297→      })
   298→      .catch(() => {});
   299→  }, []);
   300→
   301→  // ── Dialog : Besoin d'aide ? ──
   302→  const [showHelpDialog, setShowHelpDialog] = useState(false);

   310→  const [paymentPassword, setPaymentPassword] = useState('');
   311→  const [showPaymentPassword, setShowPaymentPassword] = useState(false);
   312→  const [paymentLoading, setPaymentLoading] = useState(false);
   313→  const [paymentSubmitted, setPaymentSubmitted] = useState(false);
   314→  const [paymentData, setPaymentData] = useState<any>(null);
   315→  const [paymentStats, setPaymentStats] = useState<any>(null);
   316→  const [paymentLoginError, setPaymentLoginError] = useState('');

   341→  const [expertNeed, setExpertNeed] = useState('');
   342→  const [expertUrgency, setExpertUrgency] = useState('');
   343→  const [expertSubmitted, setExpertSubmitted] = useState(false);
   344→
   345→  // Générer un numéro de ticket aléatoire
   346→  const generateTicket = (prefix: string) => {
   347→    const num = Math.floor(100000 + Math.random() * 900000);
   348→    return `${prefix}-${num}`;
   349→  };
   350→
   351→  // ── Handlers ──
   352→
   353→  const handleRequestExpert = (expert: typeof SAV_EXPERTS[number]) => {
   354→    setSelectedExpertCategoryId(expert.id);
   355→    setShowExpertPanel(true);
   356→  };

   360→    setShowExpertPanel(true);
   361→  };
   362→
   363→  const handleBackFromExpertPanel = () => {
   364→    setShowExpertPanel(false);
   365→    setSelectedExpertCategoryId(undefined);
   366→  };
   367→
   368→  const handleSubmitExpert = () => {
   369→    if (!expertName.trim() || !expertPhone.trim() || !expertNeed.trim() || !expertUrgency) {
   370→      toast.error('Veuillez remplir tous les champs.');
   371→      return;
   372→    }
   373→    const ticket = generateTicket('EXP');
   374→    toast.success(
   375→      `Demande envoyée ! Ticket ${ticket}. Notre ${selectedExpert?.title.toLowerCase()} vous contactera sous 24h.`,
   376→      { duration: 6000 },

   378→    setExpertSubmitted(true);

   382→    if (!paymentPassword.trim()) {
   383→      toast.error('Veuillez entrer votre mot de passe.');
   384→      return;
   385→    }
   386→    setPaymentLoading(true);
   387→    setPaymentLoginError('');

   389→      // 1. Login
   390→      const loginRes = await fetch('/api/auth/login', {
   391→        method: 'POST',
   392→        headers: { 'Content-Type': 'application/json' },
   393→        body: JSON.stringify({ phone: paymentPhone || undefined, password: paymentPassword }),

   395→      if (!loginRes.ok) {
   396→        const err = await loginRes.json();
   397→        setPaymentLoginError(err.error || 'Erreur de connexion');
   398→        setPaymentLoading(false);
   399→        return;

   403→      // 2. Fetch payments & stats in parallel

   405→        fetch(`/api/user/payments?userId=${user.id}`),
   406→        fetch(`/api/user/stats?userId=${user.id}`),

   411→      setPaymentData(Array.isArray(paymentsData) ? paymentsData : []);

   413→      setPaymentSubmitted(true);

   463→            <Headset className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   464→          </div>
   465→          <div className="min-w-0">
   466→            <h1 className="text-base font-bold text-foreground leading-tight truncate">
   467→              Service après-vente
   468→            </h1>
   469→            <p className="text-[11px] text-muted-foreground leading-tight">
   470→              Nous sommes à votre écoute
   471→            </p>
   472→          </div>
   473→        </div>
   474→        {setIsMenuOpen && (
   475→          <Button
   476→            variant="ghost"
   477→            size="icon"
   478→            onClick={() => setIsMenuOpen(true)}
   479→            className="hover:bg-emerald-50 dark:hover:bg-emerald-950/30 h-9 w-9"
   480→          >
   481→            <Menu className="h-5 w-5 text-foreground" />
   482→          </Button>
   483→        )}
   484→      </header>
   485→
   486→      {/* Content */}

   488→        <div className="max-w-2xl mx-auto w-full px-4 pt-3 pb-4 space-y-3.5">
   489→          {/* Hero — Besoin d'aide ? */}
   490→          <motion.div
   491→            whileHover={{ scale: 1.01 }}
   492→            whileTap={{ scale: 0.99 }}
   493→          >
   494→            <Card
