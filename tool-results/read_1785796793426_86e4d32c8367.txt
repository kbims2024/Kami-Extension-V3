  1014→    const file = e.target.files?.[0];
  1015→    if (!file) return;
  1016→    if (file.size > 5 * 1024 * 1024) { toast.error('Image trop volumineuse (max 5 Mo)'); return; }
  1017→    if (!['image/jpeg', 'image/png', 'image/webp'].includes(file.type)) { toast.error('Format non supporté (JPEG, PNG, WebP)'); return; }

  1020→      const formData = new FormData();

  1022→      formData.append('userId', currentUser.id);
  1023→      const res = await fetch('/api/user/profile', { method: 'PUT', body: formData });

  1025→        const updated = await res.json();
  1026→        setCurrentUser({ ...currentUser, ...updated });
  1027→        toast.success('Photo de profil mise à jour');
  1028→      } else { toast.error('Erreur lors du téléchargement'); }
  1029→    } catch { toast.error('Erreur serveur'); }
  1030→    finally { setPhotoUploading(false); }
  1031→  };
  1032→
  1033→  const quartiersKami = ['ASSAKLA', "N'GLOH", "N'ZOKLOH", "N'GUOUAH"];
  1034→
  1035→  const handleSave = async () => {
  1036→    if (!form.name.trim()) {
  1037→      toast.error('Le nom est obligatoire');
  1038→      return;
  1039→    }
  1040→    if (!form.pseudo.trim()) {
  1041→      toast.error('Le pseudo est obligatoire');
  1042→      return;
  1043→    }
  1044→    if (currentUser?.isResident && !form.quartier) {
  1045→      toast.error('Le quartier est obligatoire pour les résidents KAMI');
  1046→      return;
  1047→    }
  1048→
  1049→    setIsLoading(true);
  1050→    try {
  1051→      const response = await fetch('/api/user/profile', {
  1052→        method: 'PUT',
  1053→        headers: { 'Content-Type': 'application/json' },
  1054→        body: JSON.stringify({
  1055→          userId: currentUser.id,
  1056→          name: form.name,
  1057→          pseudo: form.pseudo.trim(),
  1058→          phone: form.phone,
  1059→          email: form.email,
  1060→          quartier: currentUser.isResident ? form.quartier : undefined,
  1061→        }),
  1062→      });
  1063→
  1064→      if (response.ok) {
