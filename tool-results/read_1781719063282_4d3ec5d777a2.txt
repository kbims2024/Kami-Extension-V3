    38→  const [mounted, setMounted] = useState(false);
    39→
    40→  // Store state
    41→  const {
    42→    currentUser,
    43→    setCurrentUser,
    44→    logout,
    45→    currentScreen,
    46→    setCurrentScreen,
    47→    lots,
    48→    setLots,
    49→    myReservations,
    50→    setMyReservations,

    57→    setIsReservationModalOpen,
    58→    congratulationNotification,

    61→
    62→  // Form states
    63→  const [loginName, setLoginName] = useState('');
    64→  const [loginPhone, setLoginPhone] = useState('');
    65→  const [loginIsResident, setLoginIsResident] = useState(true);
    66→  const [paymentAmount, setPaymentAmount] = useState('');

    72→      const response = await fetch('/api/lots');
    73→      if (response.ok) {
    74→        const data = await response.json();

    83→        { id: '3', name: 'B-01', surface: '400m²', priceRes: 150000, priceNon: 200000, status: 'AVAILABLE' },
    84→      ]);
    85→    }
    86→  };
    87→
    88→  const loadMyReservations = async () => {
    89→    try {
    90→      const response = await fetch(`/api/reservations?userId=${currentUser?.id}`);
    91→      if (response.ok) {
    92→        const data = await response.json();
    93→        setMyReservations(data);
    94→      }
    95→    } catch (error) {
    96→      console.error('Error loading reservations:', error);
    97→    }
    98→  };

   100→  const checkCongratulationNotifications = async () => {

   104→      const response = await fetch(`/api/user/notifications?userId=${currentUser?.id}`);
   105→      if (response.ok) {
   106→        const data = await response.json();

   108→          setCongratulationNotification({
   109→            show: true,
   110→            lotName: data.lotName,
   111→            lotBlock: data.lotBlock,

   116→      console.error('Error checking notifications:', error);
   117→    }
   118→  };

   131→      const response = await fetch('/api/auth/login', {
   132→        method: 'POST',
   133→        headers: { 'Content-Type': 'application/json' },
   134→        body: JSON.stringify({
   135→          name,
   136→          phone,
   137→          password: password || undefined,
   138→        }),
   139→      });
   140→
   141→      if (response.ok) {
   142→        const user = await response.json();
   143→        setCurrentUser(user);
   144→        toast.success(`Bienvenue ${user.name} !`);
   145→        setCurrentScreen('home');
   146→      } else {
   147→        const data = await response.json();
   148→        toast.error(data.error || 'Erreur de connexion');
   149→      }

   153→    }

   158→      toast.error('Veuillez vous connecter pour réserver');

   160→      return;

   167→
   168→  const handleReservation = async () => {

   170→
   171→    const amount = parseInt(paymentAmount);
   172→    const totalPrice = currentUser.isResident ? selectedLot.priceRes : selectedLot.priceNon;
   173→
   174→    if (!amount || amount < 10000) {
   175→      toast.error('Le montant minimum est de 10 000 FCFA');
   176→      return;
   177→    }
   178→
   179→    if (amount > totalPrice) {
   180→      toast.error('Le montant dépasse le prix du lot');
   181→      return;
   182→    }

   184→    try {
   185→      const response = await fetch('/api/reservations', {
   186→        method: 'POST',
   187→        headers: { 'Content-Type': 'application/json' },
   188→        body: JSON.stringify({
   189→          userId: currentUser.id,
   190→          lotId: selectedLot.id,
   191→          amount,
   192→          totalPrice,
   193→          isResident: currentUser.isResident,
   194→        }),
   195→      });
   196→
   197→      if (response.ok) {
   198→        const reservation = await response.json();
   199→        addReservation(reservation);
   200→        toast.success('Paiement validé avec succès !');
   201→        setIsReservationModalOpen(false);
   202→        loadLots();
   203→      } else {
   204→        toast.error('Erreur lors de la réservation');
   205→      }
   206→    } catch (error) {

   208→      const newReservation = {
   209→        id: `RES-${Date.now()}`,
   210→        lotId: selectedLot.id,
   211→        lotName: selectedLot.name,
   212→        surface: selectedLot.surface,
   213→        paidAmount: amount,
   214→        totalPrice,
   215→        isResident: currentUser.isResident,
   216→        status: amount === totalPrice ? ('PAID' as const) : ('RESERVED' as const),
   217→      };
   218→      addReservation(newReservation);
   219→      toast.success('Paiement validé avec succès !');
   220→      setIsReservationModalOpen(false);
   221→    }
   222→  };
   223→
   224→  const copyReferralLink = () => {
   225→    if (!currentUser?.referralCode) return;
   226→    navigator.clipboard.writeText(`kami.app/ref/${currentUser.referralCode}`);
   227→    toast.success('Lien copié !');
   228→  };
   229→
   230→  const handleRegistrationComplete = async (userData: { name: string; phone: string; isResident: boolean; password: string }) => {
   231→    try {
   232→      const response = await fetch('/api/auth/register', {
   233→        method: 'POST',
   234→        headers: { 'Content-Type': 'application/json' },
   235→        body: JSON.stringify({
   236→          name: userData.name,
   237→          phone: userData.phone,
   238→          isResident: userData.isResident,
   239→          password: userData.password,
   240→        }),
   241→      });
   242→
   243→      if (response.ok) {
   244→        const user = await response.json();
   245→        setCurrentUser(user);
   246→        toast.success(`Bienvenue ${user.name} ! Votre compte est créé.`);
   247→        setCurrentScreen('map');
   248→      } else {
   249→        const data = await response.json();
   250→        toast.error(data.error || 'Erreur lors de la création du compte');
   251→      }
   252→    } catch (error) {
   253→      console.error('Registration error:', error);
   254→      toast.error('Erreur lors de la création du compte');
   255→    }
   256→  };
   257→
   258→  if (!mounted) return null;
   259→
   260→  return (
   261→    <div className="min-h-screen flex flex-col bg-background">
   262→      {/* Toast Container */}
   263→      <div id="toast-container" />
   264→
   265→      {/* Modern Side Menu */}
   266→      <ModernSideMenu
   267→        isOpen={isMenuOpen}
   268→        onClose={() => setIsMenuOpen(false)}
   269→        currentUser={currentUser}
   270→        onNavigate={setCurrentScreen}
   271→        onLogout={() => {
   272→          logout();

   274→        }}
   275→      />
   276→
   277→      {/* Screens */}
   278→      {/* Page d'accueil persuasive pour tous */}

   280→        <PageTransition>
   281→          <PersuasiveLandingPage

   287→                setCurrentScreen('auth-choice');
   288→              }

   292→          />

   294→      )}
   295→
   296→      {/* Écran d'inscription en 2 étapes */}

   299→          <TwoStepRegistration
   300→            onComplete={handleRegistrationComplete}
   301→            onBack={() => setCurrentScreen('home')}

   303→          />
   304→        </PageTransition>
   305→      )}
   306→
   307→      {/* Écran de choix d'authentification */}
   308→      {!currentUser && currentScreen === 'auth-choice' && (
   309→        <PageTransition>
   310→          <AuthChoiceScreen
   311→            onLoginClick={() => setCurrentScreen('login-screen')}
   312→            onRegisterClick={() => setCurrentScreen('register')}
   313→            onBack={() => setCurrentScreen('home')}
   314→            setIsMenuOpen={setIsMenuOpen}
   315→          />
   316→        </PageTransition>
   317→      )}
   318→
   319→      {/* Écran de connexion */}

   322→          <LoginScreen
   323→            onLogin={(name, phone) => handleLogin(name, phone)}

   326→          />
   327→        </PageTransition>
   328→      )}
   329→
   330→      {/* Écran de connexion (ancien, pour compatibilité) */}
   331→      {!currentUser && currentScreen === 'login' && (
   332→        <PageTransition>
   333→          <AuthChoiceScreen
   334→            onLoginClick={() => setCurrentScreen('login-screen')}
   335→            onRegisterClick={() => setCurrentScreen('register')}
   336→            onBack={() => setCurrentScreen('home')}
   337→            setIsMenuOpen={setIsMenuOpen}
   338→          />
   339→        </PageTransition>
   340→      )}
   341→
   342→      {currentScreen === 'map' && (
   343→        <PageTransition>
   344→          <EnhancedMapScreen
   345→            lots={lots}
   346→            handleOpenReservation={handleOpenReservation}
   347→            setCurrentScreen={setCurrentScreen}
   348→            setIsMenuOpen={setIsMenuOpen}
   349→          />
   350→        </PageTransition>
   351→      )}
   352→
   353→      {currentScreen === 'dashboard' && (
   354→        <PageTransition>
   355→          <UserDashboard
   356→            currentUser={currentUser}

   361→      )}
   362→
   363→      {currentScreen === 'profile' && (

   373→

   392→      )}
   393→
   394→      {currentScreen === 'admin' && (
   395→        <PageTransition>

   400→            loadLots={loadLots}
   401→            setCurrentScreen={setCurrentScreen}
   402→          />
   403→        </PageTransition>
   404→      )}
   405→
   406→      {currentScreen === 'admin-flash-infos' && (

   408→          <FlashInfoAdmin onBack={() => setCurrentScreen('home')} />
   409→        </PageTransition>
   410→      )}
   411→
   412→      {currentScreen === 'settings' && (
   413→        <PageTransition>
   414→          <SettingsPage onBack={() => setCurrentScreen('home')} />
   415→        </PageTransition>
   416→      )}
   417→
   418→      {currentScreen === 'plan' && (
   419→        <PageTransition>
   420→          <PlanPage
   421→            setCurrentScreen={setCurrentScreen}
   422→            setIsMenuOpen={setIsMenuOpen}
   423→          />
   424→        </PageTransition>
   425→      )}
   426→
   427→      {currentScreen === 'chat' && (
   428→        <PageTransition>
   429→          <ChatPage
   430→            setCurrentScreen={setCurrentScreen}
   431→            setIsMenuOpen={setIsMenuOpen}
   432→          />
   433→        </PageTransition>

   435→
   436→      {currentScreen === 'admin-chat' && (
   437→        <PageTransition>

   444→

   448→        </PageTransition>

   450→
   451→      {/* Reservation Modal */}
   452→      <Dialog open={isReservationModalOpen} onOpenChange={setIsReservationModalOpen}>
   453→        <DialogContent className="max-w-lg rounded-t-3xl">

   455→          <DialogHeader>

   457→            <p className="text-xs text-muted-foreground">{selectedLot?.surface}</p>

   460→          {selectedLot && currentUser && (
   461→            <div className="space-y-4">

   470→                      </p>
   471→                    </div>
   472→                  </div>
   473→                  <div className="text-right">
   474→                    <p className="text-xs text-muted-foreground">Prix appliqué</p>
   475→                    <p className="font-extrabold text-[#8B5E3C]">
   476→                      {(currentUser.isResident ? selectedLot.priceRes : selectedLot.priceNon).toLocaleString('fr-FR')} F

   486→                  type="number"

   489→                  value={paymentAmount}
   490→                  onChange={(e) => setPaymentAmount(e.target.value)}
   491→                />
   492→              </div>
   493→
   494→              <Card className="bg-orange-50 border-orange-200 dark:bg-orange-900/20 dark:border-orange-800">
   495→                <CardContent className="p-3">
   496→                  <label className="flex items-center space-x-2 cursor-pointer">
   497→                    <Checkbox
   498→                      checked={agreeRules}
   499→                      onCheckedChange={(checked) => setAgreeRules(checked as boolean)}
   500→                    />
   501→                    <span className="text-xs font-bold text-foreground">
   502→                      J'ACCEPTE LE RÈGLEMENT INTÉRIEUR
   503→                    </span>
   504→                  </label>
   505→                </CardContent>
   506→              </Card>
   507→
   508→              <DialogFooter className="flex-col gap-2">

   510→                  className="w-full bg-[#10B981] hover:bg-[#059669] text-white font-bold py-4 rounded-2xl"
   511→                  disabled={!agreeRules}
   512→                  onClick={handleReservation}
   513→                >
   514→                  Confirmer le paiement
   515→                </Button>

   517→                  variant="ghost"
   518→                  className="w-full text-muted-foreground"
   519→                  onClick={() => setIsReservationModalOpen(false)}
   520→                >
   521→                  Annuler
   522→                </Button>
   523→              </DialogFooter>
   524→            </div>
   525→          )}
   526→        </DialogContent>
   527→      </Dialog>
   528→
   529→
   530→      {/* Footer */}
   531→      <footer className="mt-auto bg-card border-t border-border py-4 px-6 text-center text-sm text-muted-foreground">
   532→        <p>© 2024 KAMI-EXTENSION - Tous droits réservés</p>
   533→      </footer>
   534→
   535→      {/* Congratulation Notification */}
   536→      <CongratulationNotification />
   537→    </div>
   538→  );
   539→}
