   384→   384→// ── Fallback avatar with initials ──
   385→   385→function ExpertAvatar({ expert, size = 'lg' }: { expert: Expert; size?: 'sm' | 'lg' }) {
   386→   386→  const [imgError, setImgError] = useState(false);
   387→   387→  const initials = expert.name
   388→   388→    .split(' ')
   389→   389→    .map((n) => n[0])
   390→   390→    .join('')
   391→   391→    .slice(0, 2)
   392→   392→    .toUpperCase();
   393→   393→
   394→   394→  const sizeClass = size === 'lg' ? 'w-20 h-20 text-2xl' : 'w-12 h-12 text-sm';
   395→   395→
   396→   396→  if (imgError || !expert.image) {
   397→   397→    return (
   398→   398→      <div
   399→   399→        className={`${sizeClass} rounded-full flex items-center justify-center font-bold text-white shrink-0 shadow-lg`}
   400→   400→        style={{ backgroundColor: EXPERT_CATEGORIES.find((c) => c.experts.some((e) => e.id === expert.id))?.color || '#6B7280' }}
   401→   401→      >
   402→   402→        {initials}
   403→   403→      </div>
   404→   404→    );
   405→   405→  }
   406→   406→
   407→   407→  return (
   408→   408→    <img
   409→   409→      src={expert.image}
   410→   410→      alt={expert.name}
   411→   411→      className={`${sizeClass} rounded-full object-cover shrink-0 shadow-lg border-2 border-white/50`}
   412→   412→      onError={() => setImgError(true)}
   413→   413→    />
   414→   414→  );
   415→   415→}
