    90→    try {

    93→        const data = await response.json();

   143→        const user = await response.json();

   145→        toast.success(`Bienvenue ${user.name} !`);

   148→        const data = await response.json();

   190→          userId: currentUser.id,
   191→          lotId: selectedLot.id,
   192→          amount,
   193→          totalPrice,
   194→          isResident: currentUser.isResident,
   195→        }),

   202→        setIsReservationModalOpen(false);

   205→        toast.error('Erreur lors de la réservation');

   212→        lotName: selectedLot.name,
   213→        surface: selectedLot.surface,
   214→        paidAmount: amount,
   215→        totalPrice,
   216→        isResident: currentUser.isResident,
   217→        status: amount === totalPrice ? ('PAID' as const) : ('RESERVED' as const),
   218→      };
   219→      addReservation(newReservation);
   220→      toast.success('Paiement validé avec succès !');
   221→      setIsReservationModalOpen(false);
   222→    }
   223→  };
   224→

   237→          name: userData.name,
   238→          phone: userData.phone,
   239→          isResident: userData.isResident,
   240→          password: userData.password,
   241→        }),
   242→      });

   244→      if (response.ok) {
   245→        const user = await response.json();

   247→        toast.success(`Bienvenue ${user.name} ! Votre compte est créé.`);

   254→      console.error('Registration error:', error);
   255→      toast.error('Erreur lors de la création du compte');
   256→    }
   257→  };
   258→
   259→  if (!mounted) return null;
   260→

   316→          />

   328→        </PageTransition>
   329→      )}
   330→
   331→      {/* Écran de connexion (ancien, pour compatibilité) */}

   342→
   343→      {currentScreen === 'map' && (

   394→

   410→        </PageTransition>
   411→      )}
   412→
   413→      {currentScreen === 'settings' && (
   414→        <PageTransition>
   415→          <SettingsPage onBack={() => setCurrentScreen('home')} />
   416→        </PageTransition>
   417→      )}
   418→
   419→      {currentScreen === 'plan' && (
   420→        <PageTransition>
   421→          <PlanPage
   422→            setCurrentScreen={setCurrentScreen}
   423→            setIsMenuOpen={setIsMenuOpen}
   424→          />
   425→        </PageTransition>
   426→      )}
   427→
   428→      {currentScreen === 'chat' && (
   429→        <PageTransition>
   430→          <ChatPage

   432→            setIsMenuOpen={setIsMenuOpen}

   471→                      </p>

   474→                  <div className="text-right">
   475→                    <p className="text-xs text-muted-foreground">Prix appliqué</p>
   476→                    <p className="font-extrabold text-[#8B5E3C]">
   477→                      {(currentUser.isResident ? selectedLot.priceRes : selectedLot.priceNon).toLocaleString('fr-FR')} F

   494→
   495→              <Card className="bg-orange-50 border-orange-200 dark:bg-orange-900/20 dark:border-orange-800">
   496→                <CardContent className="p-3">
   497→                  <label className="flex items-center space-x-2 cursor-pointer">

   501→                    />
   502→                    <span className="text-xs font-bold text-foreground">
   503→                      J'ACCEPTE LE RÈGLEMENT INTÉRIEUR
   504→                    </span>
   505→                  </label>
   506→                </CardContent>
   507→              </Card>

   509→              <DialogFooter className="flex-col gap-2">

   512→                  disabled={!agreeRules}
   513→                  onClick={handleReservation}
   514→                >
   515→                  Confirmer le paiement

   519→                  className="w-full text-muted-foreground"
   520→                  onClick={() => setIsReservationModalOpen(false)}
   521→                >
   522→                  Annuler
   523→                </Button>
   524→              </DialogFooter>
   525→            </div>
   526→          )}
   527→        </DialogContent>
   528→      </Dialog>
   529→
   530→
   531→      {/* Footer */}
   532→      <footer className="mt-auto bg-card border-t border-border py-4 px-6 text-center text-sm text-muted-foreground">
   533→        <p>© 2024 KAMI-EXTENSION - Tous droits réservés</p>
   534→      </footer>
   535→
   536→      {/* Congratulation Notification */}
   537→      <CongratulationNotification />
   538→    </div>
   539→  );
   540→}
