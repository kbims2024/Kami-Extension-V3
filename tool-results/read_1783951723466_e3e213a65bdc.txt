    39→export default function KamiExtensionPage() {
    40→  const [mounted, setMounted] = useState(false);
    41→
    42→  // Store state
    43→  const {
    44→    currentUser,
    45→    setCurrentUser,
    46→    logout,
    47→    currentScreen,
    48→    setCurrentScreen,
    49→    lots,
    50→    setLots,
    51→    myReservations,
    52→    setMyReservations,
    53→    addReservation,
    54→    isMenuOpen,
    55→    setIsMenuOpen,
    56→    selectedLot,
    57→    setSelectedLot,
    58→    isReservationModalOpen,
    59→    setIsReservationModalOpen,
    60→    congratulationNotification,

    62→  } = useAppStore();
    63→
    64→  // Form states
    65→  const [loginName, setLoginName] = useState('');
    66→  const [loginPhone, setLoginPhone] = useState('');
    67→  const [loginIsResident, setLoginIsResident] = useState(true);
    68→  const [paymentAmount, setPaymentAmount] = useState('');
    69→  const [agreeRules, setAgreeRules] = useState(false);
    70→  const [adminView, setAdminView] = useState<string | null>(null);
    71→
    72→  const loadLots = async () => {
    73→    try {
    74→      const response = await fetch('/api/lots');
    75→      if (response.ok) {
    76→        const data = await response.json();
    77→        setLots(data);
    78→      }
    79→    } catch (error) {
    80→      console.error('Error loading lots:', error);
    81→      // Use demo data if API fails
    82→      setLots([
    83→        { id: '1', name: 'A-01', surface: '300m²', priceRes: 100000, priceNon: 150000, status: 'AVAILABLE' },
    84→        { id: '2', name: 'A-02', surface: '350m²', priceRes: 100000, priceNon: 150000, status: 'AVAILABLE' },
    85→        { id: '3', name: 'B-01', surface: '400m²', priceRes: 150000, priceNon: 200000, status: 'AVAILABLE' },
    86→      ]);
    87→    }
    88→  };
    89→
    90→  const loadMyReservations = async () => {
    91→    try {
    92→      const response = await fetch(`/api/reservations?userId=${currentUser?.id}`);
    93→      if (response.ok) {
    94→        const data = await response.json();
    95→        setMyReservations(data);
    96→      }
    97→    } catch (error) {
    98→      console.error('Error loading reservations:', error);
    99→    }
   100→  };
   101→
   102→  const checkCongratulationNotifications = async () => {
   103→    if (!currentUser?.id) return;
   104→
   105→    try {
   106→      const response = await fetch(`/api/user/notifications?userId=${currentUser?.id}`);
   107→      if (response.ok) {
   108→        const data = await response.json();

   110→          setCongratulationNotification({
   111→            show: true,
   112→            lotName: data.lotName,
   113→            lotBlock: data.lotBlock,
   114→          });
   115→        }
   116→      }
   117→    } catch (error) {
   118→      console.error('Error checking notifications:', error);
   119→    }
   120→  };
   121→
   122→  useEffect(() => {
   123→    setMounted(true);
   124→    loadLots();

   126→      loadMyReservations();
   127→      checkCongratulationNotifications();
   128→    }
   129→  }, [currentUser]);
   130→

   132→    try {
   133→      const response = await fetch('/api/auth/login', {
   134→        method: 'POST',
   135→        headers: { 'Content-Type': 'application/json' },
   136→        body: JSON.stringify({
   137→          name,
   138→          phone,
   139→          password: password || undefined,
   140→        }),
   141→      });
   142→
   143→      if (response.ok) {
   144→        const user = await response.json();
   145→        setCurrentUser(user);
   146→        toast.success(`Bienvenue ${user.name} !`);
   147→        setCurrentScreen('home');
   148→      } else {
   149→        const data = await response.json();
   150→        toast.error(data.error || 'Erreur de connexion');
   151→      }
   152→    } catch (error) {
   153→      console.error('Login error:', error);
   154→      toast.error('Erreur de connexion');
   155→    }
   156→  };
   157→
   158→  const handleOpenReservation = (lot: typeof lots[0]) => {
   159→    if (!currentUser) {
   160→      toast.error('Veuillez vous connecter pour réserver');
   161→      setCurrentScreen('auth-choice');
   162→      return;
   163→    }
   164→    setSelectedLot(lot);
   165→    setPaymentAmount('');
   166→    setAgreeRules(false);
   167→    setIsReservationModalOpen(true);
   168→  };
   169→
   170→  const handleReservation = async () => {
   171→    if (!selectedLot || !currentUser) return;
   172→
   173→    const amount = parseInt(paymentAmount);
   174→    const totalPrice = currentUser.isResident ? selectedLot.priceRes : selectedLot.priceNon;
   175→
   176→    if (!amount || amount < 10000) {
   177→      toast.error('Le montant minimum est de 10 000 FCFA');
   178→      return;
   179→    }
   180→
   181→    if (amount > totalPrice) {
   182→      toast.error('Le montant dépasse le prix du lot');
   183→      return;
   184→    }
   185→
   186→    try {
   187→      const response = await fetch('/api/reservations', {
   188→        method: 'POST',
   189→        headers: { 'Content-Type': 'application/json' },
   190→        body: JSON.stringify({
   191→          userId: currentUser.id,
   192→          lotId: selectedLot.id,
   193→          amount,
   194→          totalPrice,
   195→          isResident: currentUser.isResident,
   196→        }),
   197→      });
   198→
   199→      if (response.ok) {
   200→        const reservation = await response.json();
   201→        addReservation(reservation);
   202→        toast.success('Paiement validé avec succès !');
   203→        setIsReservationModalOpen(false);
   204→        loadLots();
   205→      } else {
   206→        toast.error('Erreur lors de la réservation');
   207→      }
   208→    } catch (error) {
   209→      // Fallback for demo
   210→      const newReservation = {
   211→        id: `RES-${Date.now()}`,
   212→        lotId: selectedLot.id,
   213→        lotName: selectedLot.name,
   214→        surface: selectedLot.surface,
   215→        paidAmount: amount,
   216→        totalPrice,
   217→        isResident: currentUser.isResident,
   218→        status: amount === totalPrice ? ('PAID' as const) : ('RESERVED' as const),
   219→      };
   220→      addReservation(newReservation);
   221→      toast.success('Paiement validé avec succès !');
   222→      setIsReservationModalOpen(false);
   223→    }
   224→  };
   225→
   226→  const copyReferralLink = () => {
   227→    if (!currentUser?.referralCode) return;
   228→    navigator.clipboard.writeText(`kami.app/ref/${currentUser.referralCode}`);
   229→    toast.success('Lien copié !');
   230→  };
   231→
   232→  const handleRegistrationComplete = async (userData: { name: string; pseudo: string; phone: string; isResident: boolean; password: string; quartier?: string }) => {
   233→    try {
   234→      const response = await fetch('/api/auth/register', {
   235→        method: 'POST',
   236→        headers: { 'Content-Type': 'application/json' },
   237→        body: JSON.stringify({
   238→          name: userData.name,
   239→          pseudo: userData.pseudo,
   240→          phone: userData.phone,
   241→          isResident: userData.isResident,
   242→          password: userData.password,
   243→          quartier: userData.quartier,
   244→        }),
   245→      });
   246→
   247→      if (response.ok) {
   248→        const user = await response.json();
   249→        setCurrentUser(user);
   250→        toast.success(`Bienvenue ${user.pseudo || user.name} ! Votre compte est créé.`);
   251→        setCurrentScreen('map');
   252→      } else {
   253→        const data = await response.json();
   254→        toast.error(data.error || 'Erreur lors de la création du compte');
   255→      }
   256→    } catch (error) {
   257→      console.error('Registration error:', error);
   258→      toast.error('Erreur lors de la création du compte');
   259→    }
   260→  };
   261→
   262→  if (!mounted) return null;
   263→
   264→  return (
   265→    <div className="min-h-screen flex flex-col bg-background">
   266→      {/* Toast Container */}
   267→      <div id="toast-container" />
   268→
   269→      {/* Modern Side Menu */}
   270→      <ModernSideMenu
   271→        isOpen={isMenuOpen}
   272→        onClose={() => setIsMenuOpen(false)}
   273→        currentUser={currentUser}
   274→        onNavigate={setCurrentScreen}
   275→        onLogout={() => {
   276→          logout();
   277→          toast.success('Déconnexion réussie');
   278→        }}
   279→      />
   280→
   281→      {/* Screens */}
   282→      {/* Page d'accueil persuasive pour tous */}
   283→      {currentScreen === 'home' && (
   284→        <PageTransition>
   285→          <PersuasiveLandingPage
   286→            lots={lots}
   287→            onReserveClick={() => {
   288→              if (currentUser) {
   289→                setCurrentScreen('map');
   290→              } else {
   291→                setCurrentScreen('auth-choice');
   292→              }
   293→            }}
   294→            setIsMenuOpen={setIsMenuOpen}
   295→            setCurrentScreen={setCurrentScreen}
   296→          />
   297→        </PageTransition>
   298→      )}
   299→
   300→      {/* Écran d'inscription en 2 étapes */}

   302→        <PageTransition>
   303→          <TwoStepRegistration
   304→            onComplete={handleRegistrationComplete}
   305→            onBack={() => setCurrentScreen('home')}
   306→            setIsMenuOpen={setIsMenuOpen}
   307→          />
   308→        </PageTransition>
   309→      )}
   310→
   311→      {/* Écran de choix d'authentification */}
   312→      {!currentUser && currentScreen === 'auth-choice' && (
   313→        <PageTransition>
   314→          <AuthChoiceScreen
   315→            onLoginClick={() => setCurrentScreen('login-screen')}
   316→            onRegisterClick={() => setCurrentScreen('register')}
   317→            onBack={() => setCurrentScreen('home')}
   318→            setIsMenuOpen={setIsMenuOpen}
   319→          />
   320→        </PageTransition>
   321→      )}
   322→
   323→      {/* Écran de connexion */}
   324→      {!currentUser && currentScreen === 'login-screen' && (
   325→        <PageTransition>
   326→          <LoginScreen
   327→            onLogin={(name, phone) => handleLogin(name, phone)}
   328→            onBack={() => setCurrentScreen('auth-choice')}
   329→            setIsMenuOpen={setIsMenuOpen}
   330→          />
   331→        </PageTransition>
   332→      )}
   333→
   334→      {/* Écran de connexion (ancien, pour compatibilité) */}
   335→      {!currentUser && currentScreen === 'login' && (
   336→        <PageTransition>
   337→          <AuthChoiceScreen
   338→            onLoginClick={() => setCurrentScreen('login-screen')}
   339→            onRegisterClick={() => setCurrentScreen('register')}
   340→            onBack={() => setCurrentScreen('home')}
   341→            setIsMenuOpen={setIsMenuOpen}
   342→          />
   343→        </PageTransition>
   344→      )}
   345→
   346→      {currentScreen === 'map' && (
   347→        <PageTransition>
   348→          <EnhancedMapScreen
   349→            lots={lots}
   350→            handleOpenReservation={handleOpenReservation}

   352→            setIsMenuOpen={setIsMenuOpen}
   353→          />
   354→        </PageTransition>
   355→      )}
   356→
   357→      {currentScreen === 'dashboard' && (
   358→        <PageTransition>
   359→          <UserDashboard
   360→            currentUser={currentUser}
   361→            setCurrentScreen={setCurrentScreen}
   362→            setIsMenuOpen={setIsMenuOpen}
   363→          />
   364→        </PageTransition>
   365→      )}
   366→
   367→      {currentScreen === 'profile' && (
   368→        <PageTransition>
   369→          <ProfileScreen
   370→            currentUser={currentUser}
   371→            setCurrentUser={setCurrentUser}
   372→            copyReferralLink={copyReferralLink}
   373→            setCurrentScreen={setCurrentScreen}
   374→            setIsMenuOpen={setIsMenuOpen}
   375→          />
   376→        </PageTransition>
   377→      )}
   378→
   379→      {currentScreen === 'affiliation' && (
   380→        <PageTransition>
   381→          <AffiliationScreen
   382→            currentUser={currentUser}
   383→            copyReferralLink={copyReferralLink}
   384→            setCurrentScreen={setCurrentScreen}
   385→            setIsMenuOpen={setIsMenuOpen}
   386→          />
   387→        </PageTransition>
   388→      )}
   389→
   390→      {currentScreen === 'rules' && (
   391→        <PageTransition>
   392→          <RulesScreen
   393→            setCurrentScreen={setCurrentScreen}
   394→            setIsMenuOpen={setIsMenuOpen}
   395→          />
   396→        </PageTransition>
   397→      )}
   398→
   399→      {currentScreen === 'admin' && (
   400→        <PageTransition>
   401→          <AdminScreen
   402→            adminView={adminView}
   403→            setAdminView={setAdminView}
   404→            lots={lots}
   405→            loadLots={loadLots}
   406→            setCurrentScreen={setCurrentScreen}
   407→          />
   408→        </PageTransition>
   409→      )}
   410→
   411→      {currentScreen === 'admin-flash-infos' && (
   412→        <PageTransition>
   413→          <FlashInfoAdmin onBack={() => setCurrentScreen('home')} />
   414→        </PageTransition>
   415→      )}
   416→
   417→      {currentScreen === 'settings' && (
   418→        <PageTransition>
   419→          <SettingsPage onBack={() => setCurrentScreen('home')} />
   420→        </PageTransition>
   421→      )}
   422→
   423→      {currentScreen === 'plan' && (
   424→        <PageTransition>
   425→          <PlanPage
   426→            setCurrentScreen={setCurrentScreen}
   427→            setIsMenuOpen={setIsMenuOpen}
   428→          />
   429→        </PageTransition>
   430→      )}
   431→
   432→      {currentScreen === 'chat' && (
   433→        <PageTransition>
   434→          <ChatPage
   435→            setCurrentScreen={setCurrentScreen}
   436→            setIsMenuOpen={setIsMenuOpen}
   437→          />
   438→        </PageTransition>
   439→      )}
   440→
   441→      {currentScreen === 'admin-chat' && (
   442→        <PageTransition>
   443→          <AdminChatPage
   444→            setCurrentScreen={setCurrentScreen}
   445→            setIsMenuOpen={setIsMenuOpen}
   446→          />
   447→        </PageTransition>
   448→      )}
   449→
   450→      {currentScreen === 'management-committee' && (
   451→        <PageTransition>
   452→          <ManagementCommitteeManagement onBack={() => setCurrentScreen('home')} setCurrentScreen={setCurrentScreen} />
   453→        </PageTransition>
   454→      )}
   455→
   456→      {/* Reservation Modal */}
   457→      <Dialog open={isReservationModalOpen} onOpenChange={setIsReservationModalOpen}>
   458→        <DialogContent className="max-w-lg rounded-t-3xl">
   459→          <div className="w-10 h-1 bg-muted rounded-full mx-auto mb-4" />
   460→          <DialogHeader>
   461→            <DialogTitle className="text-2xl font-bold">Lot {selectedLot?.name}</DialogTitle>
   462→            <p className="text-xs text-muted-foreground">{selectedLot?.surface}</p>
   463→          </DialogHeader>
   464→
   465→          {selectedLot && currentUser && (
   466→            <div className="space-y-4">
   467→              <Card className="bg-card border-border">
   468→                <CardContent className="p-3 flex items-center justify-between">
   469→                  <div className="flex items-center">
   470→                    <User className="mr-3 h-5 w-5 text-[#8B5E3C]" />
   471→                    <div>
   472→                      <p className="text-xs text-muted-foreground">Votre statut</p>
   473→                      <p className="font-bold text-foreground">
   474→                        {currentUser.isResident ? 'Résident KAMI' : 'Non-Résident'}
   475→                      </p>
   476→                    </div>
   477→                  </div>
   478→                  <div className="text-right">
   479→                    <p className="text-xs text-muted-foreground">Prix appliqué</p>
   480→                    <p className="font-extrabold text-[#8B5E3C]">
   481→                      {(currentUser.isResident ? selectedLot.priceRes : selectedLot.priceNon).toLocaleString('fr-FR')} F
   482→                    </p>
   483→                  </div>
   484→                </CardContent>
   485→              </Card>
   486→
   487→              <div>
   488→                <Label>Montant à payer maintenant</Label>
   489→                <p className="text-xs text-muted-foreground mb-2">Minimum requis : 10 000 F</p>
   490→                <Input
   491→                  type="number"
   492→                  min={10000}
   493→                  placeholder="Ex: 50000"
   494→                  value={paymentAmount}
   495→                  onChange={(e) => setPaymentAmount(e.target.value)}
   496→                />
   497→              </div>
   498→
   499→              <Card className="bg-orange-50 border-orange-200 dark:bg-orange-900/20 dark:border-orange-800">
   500→                <CardContent className="p-3">
   501→                  <label className="flex items-center space-x-2 cursor-pointer">
   502→                    <Checkbox
   503→                      checked={agreeRules}
   504→                      onCheckedChange={(checked) => setAgreeRules(checked as boolean)}
   505→                    />
   506→                    <span className="text-xs font-bold text-foreground">
   507→                      J'ACCEPTE LE RÈGLEMENT INTÉRIEUR
   508→                    </span>
   509→                  </label>
   510→                </CardContent>
   511→              </Card>
   512→
   513→              <DialogFooter className="flex-col gap-2">
   514→                <Button
   515→                  className="w-full bg-[#10B981] hover:bg-[#059669] text-white font-bold py-4 rounded-2xl"
   516→                  disabled={!agreeRules}
   517→                  onClick={handleReservation}
   518→                >
   519→                  Confirmer le paiement
   520→                </Button>
   521→                <Button
   522→                  variant="ghost"
   523→                  className="w-full text-muted-foreground"
   524→                  onClick={() => setIsReservationModalOpen(false)}
   525→                >
   526→                  Annuler
   527→                </Button>
   528→              </DialogFooter>
   529→            </div>
   530→          )}
   531→        </DialogContent>
   532→      </Dialog>
   533→
   534→
   535→      {/* Footer */}
   536→      <footer className="mt-auto bg-card border-t border-border py-4 px-6 text-center text-sm text-muted-foreground">
   537→        <p>© 2024 KAMI-EXTENSION - Tous droits réservés</p>
   538→      </footer>
   539→
   540→      {/* Congratulation Notification */}
   541→      <CongratulationNotification />
   542→    </div>
   543→  );
   544→}
