   212→    if (amount > totalPrice) {
   213→      toast.error('Le montant dépasse le prix du lot');

   226→          isResident: currentUser.isResident,
   227→        }),

   237→        toast.error('Erreur lors de la réservation');

   244→        lotName: selectedLot.name,

  1034→  const quartiersKami = ['ASSAKLA', "N'GLOH", "N'ZOKLOH", "N'GUOUAH"];
  1035→
  1036→  const handleSave = async () => {
  1037→    if (!form.name.trim()) {
  1038→      toast.error('Le nom est obligatoire');
  1039→      return;
  1040→    }
  1041→    if (!form.pseudo.trim()) {
  1042→      toast.error('Le pseudo est obligatoire');
  1043→      return;
  1044→    }
  1045→    if (currentUser?.isResident && !form.quartier) {
  1046→      toast.error('Le quartier est obligatoire pour les résidents KAMI');
  1047→      return;
  1048→    }
  1049→
  1050→    setIsLoading(true);
  1051→    try {
  1052→      const response = await fetch('/api/user/profile', {
  1053→        method: 'PUT',
  1054→        headers: { 'Content-Type': 'application/json' },
  1055→        body: JSON.stringify({
  1056→          userId: currentUser.id,
  1057→          name: form.name,
  1058→          pseudo: form.pseudo.trim(),
  1059→          phone: form.phone,
  1060→          email: form.email,
  1061→          quartier: currentUser.isResident ? form.quartier : undefined,
  1062→        }),
  1063→      });
