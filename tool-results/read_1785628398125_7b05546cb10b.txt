  1011→  const handlePhotoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
  1012→    const file = e.target.files?.[0];
  1013→    if (!file) return;
  1014→    if (file.size > 5 * 1024 * 1024) { toast.error('Image trop volumineuse (max 5 Mo)'); return; }
  1015→    if (!['image/jpeg', 'image/png', 'image/webp'].includes(file.type)) { toast.error('Format non supporté (JPEG, PNG, WebP)'); return; }
  1016→    setPhotoUploading(true);

  1018→      const formData = new FormData();

  1020→      formData.append('userId', currentUser.id);
  1021→      const res = await fetch('/api/user/profile', { method: 'PUT', body: formData });

  1023→        const updated = await res.json();
  1024→        setCurrentUser({ ...currentUser, ...updated });
  1025→        toast.success('Photo de profil mise à jour');
  1026→      } else { toast.error('Erreur lors du téléchargement'); }
  1027→    } catch { toast.error('Erreur serveur'); }
  1028→    finally { setPhotoUploading(false); }
  1029→  };
  1030→
  1031→  const quartiersKami = ['ASSAKLA', "N'GLOH", "N'ZOKLOH", "N'GUOUAH"];
  1032→
  1033→  const handleSave = async () => {
  1034→    if (!form.name.trim()) {
  1035→      toast.error('Le nom est obligatoire');
  1036→      return;
  1037→    }
  1038→    if (!form.pseudo.trim()) {
  1039→      toast.error('Le pseudo est obligatoire');
  1040→      return;
  1041→    }
  1042→    if (currentUser?.isResident && !form.quartier) {
  1043→      toast.error('Le quartier est obligatoire pour les résidents KAMI');
  1044→      return;
  1045→    }
  1046→
  1047→    setIsLoading(true);
  1048→    try {
  1049→      const response = await fetch('/api/user/profile', {
  1050→        method: 'PUT',
  1051→        headers: { 'Content-Type': 'application/json' },
  1052→        body: JSON.stringify({
  1053→          userId: currentUser.id,
  1054→          name: form.name,
  1055→          pseudo: form.pseudo.trim(),
  1056→          phone: form.phone,
  1057→          email: form.email,
  1058→          quartier: currentUser.isResident ? form.quartier : undefined,
  1059→        }),
  1060→      });
  1061→
  1062→      if (response.ok) {
  1063→        const updatedUser = await response.json();
  1064→        setCurrentUser({ ...currentUser, ...updatedUser });
  1065→        toast.success('Profil mis à jour avec succès');
  1066→        setIsEditing(false);
