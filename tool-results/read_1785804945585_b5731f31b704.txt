   136→// Composant de barre de progression
   137→const ProgressBar: React.FC<{ value: number; max: number; label?: string }> = ({ value, max, label }) => {
   138→  const percentage = Math.min(100, Math.max(0, (value / max) * 100))
   139→  
   140→  return (
   141→    <div className="space-y-2">
   142→      {label && (
   143→        <motion.div 
   144→          initial={{ opacity: 0, x: -20 }}
   145→          animate={{ opacity: 1, x: 0 }}
   146→          className="flex justify-between text-sm"
   147→        >
   148→          <span className="text-muted-foreground">{label}</span>
   149→          <span className="font-medium text-foreground">{percentage.toFixed(1)}%</span>
   150→        </motion.div>
   151→      )}
   152→      <div className="h-2 bg-muted rounded-full overflow-hidden">
   153→        <motion.div
   154→          initial={{ width: 0 }}
   155→          animate={{ width: `${percentage}%` }}
   156→          transition={{ duration: 1, ease: 'easeOut' }}
   157→          className="h-full bg-gradient-to-r from-emerald-500 to-emerald-600"
   158→        />
   159→      </div>
   160→      <motion.div 
   161→        initial={{ opacity: 0 }}
   162→        animate={{ opacity: 1 }}
   163→        transition={{ delay: 0.5 }}
   164→        className="flex justify-between text-xs text-muted-foreground"
   165→      >

   167→        <span>{max.toLocaleString()} FCFA</span>
   168→      </motion.div>
   169→    </div>
   170→  )
   171→}
