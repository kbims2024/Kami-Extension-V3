   532→              </DialogFooter>
   533→            </div>

   556→          © 2024 KAMI-EXTENSION - Tous droits réservés

   563→        onOpenChange={setIsAdminLoginOpen}
   564→        onAdminLoginSuccess={(user) => {
   565→          setCurrentUser(user);
   566→          setCurrentScreen('admin');
   567→        }}
   568→      />
   569→
   570→      {/* Congratulation Notification */}
   571→      <CongratulationNotification />
   572→    </div>
   573→  );
   574→}
