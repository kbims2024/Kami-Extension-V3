    38→  const [mounted, setMounted] = useState(false);
    39→
    40→  // Store state
    41→  const {
    42→    currentUser,
    43→    setCurrentUser,
    44→    logout,
    45→    currentScreen,

    47→    lots,
    48→    setLots,
    49→    myReservations,

    61→
    62→  // Form states

    64→  const [loginPhone, setLoginPhone] = useState('');
    65→  const [loginIsResident, setLoginIsResident] = useState(true);
    66→  const [paymentAmount, setPaymentAmount] = useState('');

    72→      const response = await fetch('/api/lots');
    73→      if (response.ok) {
    74→        const data = await response.json();

    82→        { id: '2', name: 'A-02', surface: '350m²', priceRes: 100000, priceNon: 150000, status: 'AVAILABLE' },
    83→        { id: '3', name: 'B-01', surface: '400m²', priceRes: 150000, priceNon: 200000, status: 'AVAILABLE' },
    84→      ]);
    85→    }
    86→  };
    87→
    88→  const loadMyReservations = async () => {
    89→    try {
    90→      const response = await fetch(`/api/reservations?userId=${currentUser?.id}`);
    91→      if (response.ok) {
    92→        const data = await response.json();
    93→        setMyReservations(data);
    94→      }
    95→    } catch (error) {
    96→      console.error('Error loading reservations:', error);
    97→    }
    98→  };

   100→  const checkCongratulationNotifications = async () => {

   106→        const data = await response.json();

   108→          setCongratulationNotification({
   109→            show: true,
   110→            lotName: data.lotName,
   111→            lotBlock: data.lotBlock,
   112→          });
   113→        }

   116→      console.error('Error checking notifications:', error);
   117→    }
   118→  };

   131→      const response = await fetch('/api/auth/login', {
   132→        method: 'POST',
   133→        headers: { 'Content-Type': 'application/json' },
   134→        body: JSON.stringify({
   135→          name,
   136→          phone,
   137→          password: password || undefined,
   138→        }),
   139→      });

   141→      if (response.ok) {
   142→        const user = await response.json();
   143→        setCurrentUser(user);
   144→        toast.success(`Bienvenue ${user.name} !`);
   145→        setCurrentScreen('home');
   146→      } else {
   147→        const data = await response.json();
   148→        toast.error(data.error || 'Erreur de connexion');
   149→      }

   158→      toast.error('Veuillez vous connecter pour réserver');

   170→
   171→    const amount = parseInt(paymentAmount);
   172→    const totalPrice = currentUser.isResident ? selectedLot.priceRes : selectedLot.priceNon;
   173→
   174→    if (!amount || amount < 10000) {
   175→      toast.error('Le montant minimum est de 10 000 FCFA');
   176→      return;
   177→    }
   178→
   179→    if (amount > totalPrice) {
   180→      toast.error('Le montant dépasse le prix du lot');
   181→      return;

   184→    try {
   185→      const response = await fetch('/api/reservations', {
   186→        method: 'POST',
   187→        headers: { 'Content-Type': 'application/json' },
   188→        body: JSON.stringify({
   189→          userId: currentUser.id,
   190→          lotId: selectedLot.id,
   191→          amount,
   192→          totalPrice,
   193→          isResident: currentUser.isResident,
   194→        }),
   195→      });
   196→
   197→      if (response.ok) {
   198→        const reservation = await response.json();

   200→        toast.success('Paiement validé avec succès !');
   201→        setIsReservationModalOpen(false);
   202→        loadLots();
   203→      } else {
   204→        toast.error('Erreur lors de la réservation');
   205→      }
   206→    } catch (error) {

   208→      const newReservation = {
   209→        id: `RES-${Date.now()}`,
   210→        lotId: selectedLot.id,
   211→        lotName: selectedLot.name,
   212→        surface: selectedLot.surface,
   213→        paidAmount: amount,
   214→        totalPrice,
   215→        isResident: currentUser.isResident,
   216→        status: amount === totalPrice ? ('PAID' as const) : ('RESERVED' as const),
   217→      };
   218→      addReservation(newReservation);
   219→      toast.success('Paiement validé avec succès !');
   220→      setIsReservationModalOpen(false);
   221→    }
   222→  };
   223→
   224→  const copyReferralLink = () => {
   225→    if (!currentUser?.referralCode) return;
   226→    navigator.clipboard.writeText(`kami.app/ref/${currentUser.referralCode}`);
   227→    toast.success('Lien copié !');
   228→  };
   229→
   230→  const handleRegistrationComplete = async (userData: { name: string; phone: string; isResident: boolean; password: string; residenceOrigine?: string }) => {

   232→      const response = await fetch('/api/auth/login', {
   233→        method: 'POST',
   234→        headers: { 'Content-Type': 'application/json' },
   235→        body: JSON.stringify({
   236→          name: userData.name,
   237→          phone: userData.phone,
   238→          isResident: userData.isResident,
   239→          password: userData.password,
   240→          residenceOrigine: userData.residenceOrigine,
   241→        }),
   242→      });
   243→
   244→      if (response.ok) {
   245→        const user = await response.json();
   246→        setCurrentUser(user);
   247→        toast.success(`Bienvenue ${user.name} ! Votre compte est créé.`);
   248→        setCurrentScreen('map');
   249→      } else {
   250→        const data = await response.json();
   251→        toast.error(data.error || 'Erreur lors de la création du compte');
   252→      }
   253→    } catch (error) {
   254→      console.error('Registration error:', error);
   255→      toast.error('Erreur lors de la création du compte');
   256→    }
   257→  };
   258→
   259→  if (!mounted) return null;
   260→
   261→  return (
   262→    <div className="min-h-screen flex flex-col bg-background">
   263→      {/* Toast Container */}
   264→      <div id="toast-container" />
   265→
   266→      {/* Modern Side Menu */}

   268→        isOpen={isMenuOpen}
   269→        onClose={() => setIsMenuOpen(false)}
   270→        currentUser={currentUser}
   271→        onNavigate={setCurrentScreen}
   272→        onLogout={() => {
   273→          logout();

   275→        }}
   276→      />

   278→      {/* Screens */}

   293→          />

   295→      )}

   300→          <TwoStepRegistration
   301→            onComplete={handleRegistrationComplete}
   302→            onBack={() => setCurrentScreen('home')}

   304→          />
   305→        </PageTransition>
   306→      )}
   307→
   308→      {/* Écran de choix d'authentification */}

   311→          <AuthChoiceScreen
   312→            onLoginClick={() => setCurrentScreen('login-screen')}

   315→            setIsMenuOpen={setIsMenuOpen}
   316→          />
   317→        </PageTransition>
   318→      )}
   319→
   320→      {/* Écran de connexion */}

   323→          <LoginScreen

   325→            onBack={() => setCurrentScreen('auth-choice')}
   326→            setIsMenuOpen={setIsMenuOpen}
   327→          />
   328→        </PageTransition>
   329→      )}
   330→
   331→      {/* Écran de connexion (ancien, pour compatibilité) */}
   332→      {!currentUser && currentScreen === 'login' && (
   333→        <PageTransition>
   334→          <AuthChoiceScreen
   335→            onLoginClick={() => setCurrentScreen('login-screen')}
   336→            onRegisterClick={() => setCurrentScreen('register')}
   337→            onBack={() => setCurrentScreen('home')}
   338→            setIsMenuOpen={setIsMenuOpen}
   339→          />
   340→        </PageTransition>
   341→      )}
   342→
   343→      {currentScreen === 'map' && (
   344→        <PageTransition>
   345→          <EnhancedMapScreen

   352→      )}
   353→
   354→      {currentScreen === 'dashboard' && (

   356→          <UserDashboard

   363→

   374→
   375→      {currentScreen === 'affiliation' && (

   391→          />
   392→        </PageTransition>
   393→      )}
   394→
   395→      {currentScreen === 'admin' && (
   396→        <PageTransition>
   397→          <AdminScreen

   399→            setAdminView={setAdminView}
   400→            lots={lots}
   401→            loadLots={loadLots}
   402→            setCurrentScreen={setCurrentScreen}
   403→          />
   404→        </PageTransition>
   405→      )}
   406→
   407→      {currentScreen === 'admin-flash-infos' && (
   408→        <PageTransition>
   409→          <FlashInfoAdmin onBack={() => setCurrentScreen('home')} />
   410→        </PageTransition>
   411→      )}
   412→
   413→      {currentScreen === 'settings' && (
   414→        <PageTransition>
   415→          <SettingsPage onBack={() => setCurrentScreen('home')} />
   416→        </PageTransition>
   417→      )}
   418→
   419→      {currentScreen === 'plan' && (
   420→        <PageTransition>
   421→          <PlanPage
   422→            setCurrentScreen={setCurrentScreen}
   423→            setIsMenuOpen={setIsMenuOpen}
   424→          />
   425→        </PageTransition>
   426→      )}
   427→
   428→      {currentScreen === 'chat' && (
   429→        <PageTransition>
   430→          <ChatPage
   431→            setCurrentScreen={setCurrentScreen}
   432→            setIsMenuOpen={setIsMenuOpen}
   433→          />

   435→      )}
   436→
   437→      {currentScreen === 'admin-chat' && (

   460→

   470→                        {currentUser.isResident ? 'Résident KAMI' : 'Non-Résident'}
   471→                      </p>
   472→                    </div>
   473→                  </div>
   474→                  <div className="text-right">
   475→                    <p className="text-xs text-muted-foreground">Prix appliqué</p>
   476→                    <p className="font-extrabold text-[#8B5E3C]">

   490→                  value={paymentAmount}
   491→                  onChange={(e) => setPaymentAmount(e.target.value)}

   494→
   495→              <Card className="bg-orange-50 border-orange-200 dark:bg-orange-900/20 dark:border-orange-800">
   496→                <CardContent className="p-3">
   497→                  <label className="flex items-center space-x-2 cursor-pointer">
   498→                    <Checkbox
   499→                      checked={agreeRules}
   500→                      onCheckedChange={(checked) => setAgreeRules(checked as boolean)}
   501→                    />
   502→                    <span className="text-xs font-bold text-foreground">
   503→                      J'ACCEPTE LE RÈGLEMENT INTÉRIEUR

   505→                  </label>
   506→                </CardContent>
   507→              </Card>

   510→                <Button
   511→                  className="w-full bg-[#10B981] hover:bg-[#059669] text-white font-bold py-4 rounded-2xl"

   513→                  onClick={handleReservation}

   515→                  Confirmer le paiement

   518→                  variant="ghost"
   519→                  className="w-full text-muted-foreground"
   520→                  onClick={() => setIsReservationModalOpen(false)}
   521→                >
   522→                  Annuler
   523→                </Button>
   524→              </DialogFooter>
   525→            </div>
   526→          )}
   527→        </DialogContent>
   528→      </Dialog>
   529→
   530→
   531→      {/* Footer */}
   532→      <footer className="mt-auto bg-card border-t border-border py-4 px-6 text-center text-sm text-muted-foreground">
   533→        <p>© 2024 KAMI-EXTENSION - Tous droits réservés</p>
   534→      </footer>
   535→
   536→      {/* Congratulation Notification */}
   537→      <CongratulationNotification />
   538→    </div>
   539→  );
   540→}
