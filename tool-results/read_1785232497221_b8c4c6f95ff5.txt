   387→   387→// ── Fallback avatar with initials ──
   388→   388→function ExpertAvatar({ expert, size = 'lg' }: { expert: Expert; size?: 'sm' | 'lg' }) {
   389→   389→  const [imgError, setImgError] = useState(false);
   390→   390→  const initials = expert.name
   391→   391→    .split(' ')
   392→   392→    .map((n) => n[0])
   393→   393→    .join('')
   394→   394→    .slice(0, 2)
   395→   395→    .toUpperCase();
   396→   396→
   397→   397→  const sizeClass = size === 'lg' ? 'w-20 h-20 text-2xl' : 'w-12 h-12 text-sm';
   398→   398→
   399→   399→  if (imgError || !expert.image) {
   400→   400→    return (
   401→   401→      <div
   402→   402→        className={`${sizeClass} rounded-full flex items-center justify-center font-bold text-white shrink-0 shadow-lg`}
   403→   403→        style={{ backgroundColor: EXPERT_CATEGORIES.find((c) => c.experts.some((e) => e.id === expert.id))?.color || '#6B7280' }}
   404→   404→      >
   405→   405→        {initials}
   406→   406→      </div>
   407→   407→    );
   408→   408→  }
   409→   409→
   410→   410→  return (
   411→   411→    <img
   412→   412→      src={expert.image}
   413→   413→      alt={expert.name}
   414→   414→      className={`${sizeClass} rounded-full object-cover shrink-0 shadow-lg border-2 border-white/50`}
   415→   415→      onError={() => setImgError(true)}
   416→   416→    />
   417→   417→  );
   418→   418→}
