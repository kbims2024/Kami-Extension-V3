   384→// ── Fallback avatar with initials ──
   385→function ExpertAvatar({ expert, size = 'lg' }: { expert: Expert; size?: 'sm' | 'lg' }) {
   386→  const [imgError, setImgError] = useState(false);
   387→  const initials = expert.name
   388→    .split(' ')
   389→    .map((n) => n[0])
   390→    .join('')
   391→    .slice(0, 2)
   392→    .toUpperCase();
   393→
   394→  const sizeClass = size === 'lg' ? 'w-20 h-20 text-2xl' : 'w-12 h-12 text-sm';
   395→
   396→  if (imgError || !expert.image) {
   397→    return (
   398→      <div
   399→        className={`${sizeClass} rounded-full flex items-center justify-center font-bold text-white shrink-0 shadow-lg`}
   400→        style={{ backgroundColor: EXPERT_CATEGORIES.find((c) => c.experts.some((e) => e.id === expert.id))?.color || '#6B7280' }}
   401→      >
   402→        {initials}
   403→      </div>
   404→    );
   405→  }
   406→
   407→  return (
   408→    <img
   409→      src={expert.image}
   410→      alt={expert.name}
   411→      className={`${sizeClass} rounded-full object-cover shrink-0 shadow-lg border-2 border-white/50`}
   412→      onError={() => setImgError(true)}
   413→    />
   414→  );
   415→}
